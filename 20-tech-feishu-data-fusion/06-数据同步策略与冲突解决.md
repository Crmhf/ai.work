# 06 - 数据同步策略与冲突解决

> **TL;DR**：飞书场景的数据同步有三大难题——**循环依赖、并发冲突、网络分区**。实战关键是：避免循环、串行化写操作、幂等设计、最终一致性兜底。

---

## 一、三大同步难题

### 难题 1：循环依赖

业务闭环天然存在循环：

```
数据 A 变更 → 触发任务 X 创建 → 任务 X 状态变更 → 触发数据 A 更新 → ...
```

如果不加控制，会形成**无限循环**，消耗 API 配额。

### 难题 2：并发冲突

同一记录被多个进程同时更新，后写覆盖前写，导致**数据丢失**。

### 难题 3：网络分区

Webhook 丢失 + API 限流 + 服务重启，导致**短暂的数据不一致**。

---

## 二、避免循环依赖

### 策略 1：单向同步 + 变更标记

```python
# 多维表格记录的"来源标记"字段
fields = {
    "数据来源": "manual" | "auto_sync",  # 区分人工改动 vs 自动同步
    "最后同步时间": int(time.time() * 1000),
    "最后同步触发者": "task_completion"
}

async def on_task_change(self, task_guid: str, new_status: str):
    """任务状态变更时回写多维表格"""
    task = await self.get_task(task_guid)
    origin = json.loads(task.origin)
    record_id = origin.get("bitable_record_id")
    
    if not record_id:
        return
    
    # 标记为"自动同步"
    await self.update_record(record_id, {
        "任务状态": new_status,
        "进展": self._status_to_progress(new_status),
        "数据来源": "auto_sync",
        "最后同步触发者": f"task:{task_guid}",
        "最后同步时间": int(time.time() * 1000)
    })
```

**关键**：回写时设置 `数据来源 = auto_sync`。如果 Automation 监听到这个变更再次触发同步，**我们要识别出来并跳过**。

### 策略 2：去重与节流

```python
class SyncThrottler:
    """同步节流器：避免重复同步"""
    
    def __init__(self, redis_client):
        self.redis = redis_client
    
    async def should_sync(self, record_id: str, trigger: str, ttl_seconds: int = 60) -> bool:
        """
        判断是否应该同步（基于 Redis 的去重）
        
        Args:
            record_id: 记录 ID
            trigger: 触发者（如 task_guid）
            ttl_seconds: 节流时间窗口
        """
        key = f"sync_throttle:{record_id}:{trigger}"
        
        # 检查是否在节流窗口内
        exists = await self.redis.exists(key)
        if exists:
            return False
        
        # 设置节流标记
        await self.redis.setex(key, ttl_seconds, "1")
        return True

# 使用
throttler = SyncThrottler(redis_client)

async def on_task_change(self, task_guid, new_status):
    record_id = self._get_record_id(task_guid)
    
    # 节流：同一任务 60 秒内只同步一次
    if not await throttler.should_sync(record_id, f"task:{task_guid}"):
        print(f"节流跳过: {record_id}")
        return
    
    # 同步
    await self._do_sync(record_id, new_status)
```

## 三、并发冲突解决

### 冲突场景

```
时刻 T1：
  - 进程 A 读取记录（状态：doing，进展：0.5）
  - 进程 B 读取记录（状态：doing，进展：0.5）

时刻 T2：
  - 进程 A 更新为 done，进展 1.0
  - 进程 B 更新为 done，进展 0.6（过期数据）

结果：最终状态是 done，但进展是 0.6（错误）
```

### 解决方案 1：乐观锁（版本号）

```python
async def update_with_optimistic_lock(
    self,
    app_token: str,
    table_id: str,
    record_id: str,
    new_fields: Dict,
    expected_version: int
) -> bool:
    """乐观锁更新"""
    max_retries = 3
    
    for attempt in range(max_retries):
        # 1. 读取当前版本
        record = await self.get_record(app_token, table_id, record_id)
        current_version = record.get("version", 0)
        
        if current_version != expected_version:
            print(f"版本冲突: 期望 {expected_version}，实际 {current_version}")
            # 重新读取并重试
            await asyncio.sleep(0.1 * (2 ** attempt))
            continue
        
        # 2. 尝试更新（带上版本号）
        try:
            await self.update_record(
                app_token, table_id, record_id,
                {**new_fields, "version": current_version + 1}
            )
            return True
        except VersionConflictError:
            # 其他人已经更新，重试
            await asyncio.sleep(0.1 * (2 ** attempt))
            continue
    
    return False
```

### 解决方案 2：最后写入胜出 + 审计日志

```python
async def update_with_lww(
    self,
    app_token: str,
    table_id: str,
    record_id: str,
    new_fields: Dict
) -> bool:
    """最后写入胜出 + 审计"""
    new_fields["最后同步时间"] = int(time.time() * 1000)
    
    await self.update_record(app_token, table_id, record_id, new_fields)
    
    # 记录审计日志
    await self._log_audit({
        "action": "update",
        "record_id": record_id,
        "fields": new_fields,
        "timestamp": int(time.time() * 1000),
        "source": "auto_sync"
    })
    
    return True
```

### 解决方案 3：异步队列 + 串行化

```python
class WriteQueue:
    """异步写队列：同一记录串行化"""
    
    def __init__(self):
        self.queues = {}  # record_id → asyncio.Queue
    
    async def enqueue(self, record_id: str, update_fn):
        if record_id not in self.queues:
            self.queues[record_id] = asyncio.Queue()
            asyncio.create_task(self._process_queue(record_id))
        
        await self.queues[record_id].put(update_fn)
    
    async def _process_queue(self, record_id):
        queue = self.queues[record_id]
        while True:
            update_fn = await queue.get()
            try:
                await update_fn()
            except Exception as e:
                print(f"队列更新失败: {e}")
            queue.task_done()

# 使用
write_queue = WriteQueue()

async def safe_update(record_id, fields):
    await write_queue.enqueue(
        record_id,
        lambda: self._do_update(record_id, fields)
    )
```

## 四、幂等性设计

幂等性：同一操作执行多次，结果相同。

### 场景

```python
# 因为网络重试，同一更新被发送多次
await update_record(record_id, {"进展": 0.5})
# 网络超时，重试
await update_record(record_id, {"进展": 0.5})  # 重复
```

### 实现：去重 Token

```python
class IdempotencyKey:
    """幂等性 key"""
    
    def __init__(self):
        self.keys = {}  # key → result
    
    async def execute(self, key: str, fn):
        if key in self.keys:
            print(f"幂等性命中: {key}")
            return self.keys[key]
        
        result = await fn()
        self.keys[key] = result
        return result

# 使用
idemp = IdempotencyKey()

async def sync_progress(task_guid, progress):
    key = f"sync_progress:{task_guid}:{progress}"
    
    return await idemp.execute(key, lambda: self._do_sync(task_guid, progress))
```

### 实现：服务端去重

飞书多维表格本身有去重机制（基于 record_id），但要在客户端保证幂等性。

```python
async def idempotent_update(self, record_id, fields):
    """幂等更新"""
    # 1. 读取当前状态
    record = await self.get_record(record_id)
    
    # 2. 检查是否真的需要更新
    for key, new_value in fields.items():
        current_value = record.fields.get(key)
        if current_value == new_value:
            continue  # 已经是目标值，跳过
    
    # 3. 真正更新
    return await self.update_record(record_id, fields)
```

## 五、网络分区处理

### 场景

- Webhook 丢失。
- API 限流。
- 服务重启。
- 数据库故障。

### 解决：定时补偿

```python
class ReconciliationService:
    """对账服务：保证最终一致性"""
    
    async def reconcile_all(self):
        """全量对账"""
        # 1. 对账任务跟踪表
        await self._reconcile_task_table()
        
        # 2. 对账项目主表
        await self._reconcile_project_table()
        
        # 3. 对账客户反馈表
        await self._reconcile_feedback_table()
    
    async def _reconcile_task_table(self):
        """对账任务跟踪表"""
        # 1. 拉取多维表格所有"任务跟踪表"记录
        records = await self.bitable.list_all_records(TASK_TABLE_ID)
        
        for record in records:
            task_guid = record.fields.get("任务GUID")
            if not task_guid:
                continue
            
            try:
                # 2. 获取飞书任务当前状态
                task = await self.task_manager.get_task(task_guid)
                
                # 3. 比对
                bitable_status = record.fields.get("任务状态")
                feishu_status = self._map_status(task.status)
                
                if bitable_status != feishu_status:
                    # 不一致，修复
                    print(f"修复 {task_guid}: {bitable_status} → {feishu_status}")
                    await self.bitable.update_record(
                        record.record_id,
                        {
                            "任务状态": feishu_status,
                            "进展": self._status_to_progress(task.status),
                            "数据来源": "reconciliation"
                        }
                    )
            except TaskNotFoundError:
                # 任务已删除，标记多维表格记录
                await self.bitable.update_record(
                    record.record_id,
                    {
                        "任务状态": "已取消",
                        "数据来源": "reconciliation"
                    }
                )
```

## 六、实战对账策略

### 三层防御

```
第一层：实时同步（Webhook + API）
    │ 失败
    ▼
第二层：5 分钟定时补偿
    │ 失败（如 API 限流）
    ▼
第三层：每小时全量对账
    │ 失败
    ▼
人工介入 + 告警
```

### 配置示例

```python
class Scheduler:
    """调度器"""
    
    def schedule_jobs(self):
        # 每 5 分钟：增量对账
        self.scheduler.add_job(
            self.incremental_reconciliation,
            'cron',
            minute='*/5'
        )
        
        # 每小时：全量对账
        self.scheduler.add_job(
            self.full_reconciliation,
            'cron',
            minute=0
        )
        
        # 每天凌晨：清理归档
        self.scheduler.add_job(
            self.daily_cleanup,
            'cron',
            hour=2
        )
    
    async def incremental_reconciliation(self):
        """增量对账：只对账最近 10 分钟变更的记录"""
        since = int((datetime.now() - timedelta(minutes=10)).timestamp() * 1000)
        await self.reconcile.reconcile_since(since)
    
    async def full_reconciliation(self):
        """全量对账"""
        await self.reconcile.reconcile_all()
```

## 八、本章关键 takeaway

- 三大同步难题：循环依赖、并发冲突、网络分区。
- 避免循环：单向同步 + 变更标记 + 节流 + 状态机。
- 解决并发冲突：乐观锁、LWW + 审计、异步队列串行化。
- 幂等性：去重 token + 服务端去重。
- 网络分区：定时补偿（5 分钟增量 + 每小时全量）。
- 三层防御：实时同步 → 定时补偿 → 人工介入。
- 监控关键指标：冲突率、延迟、节流率、对账差异数。
